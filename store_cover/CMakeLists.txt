cmake_minimum_required(VERSION 3.16)
project(tls_fixer_store C)
option(STORE_COVER_HOST_TESTS "Build portable artwork fallback tests" OFF)
if(STORE_COVER_HOST_TESTS)
    enable_testing()
    add_executable(store_cover_tests cover.c json_stream.c tests.c)
    target_compile_options(store_cover_tests PRIVATE -std=c11 -Wall -Wextra -Werror -UNDEBUG -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(store_cover_tests PRIVATE -fsanitize=address,undefined)
    add_test(NAME store_cover_contract COMMAND store_cover_tests)
    add_executable(store_cover_fetch_tests cover.c json_stream.c http.c fetch_tests.c)
    target_compile_options(store_cover_fetch_tests PRIVATE -std=c11 -Wall -Wextra -Werror -UNDEBUG -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(store_cover_fetch_tests PRIVATE -fsanitize=address,undefined)
    add_test(NAME store_cover_http COMMAND store_cover_fetch_tests)
    add_executable(store_cover_stream_tests cover.c json_stream.c stream_tests.c)
    target_compile_options(store_cover_stream_tests PRIVATE -std=c11 -Wall -Wextra -Werror -UNDEBUG -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(store_cover_stream_tests PRIVATE -fsanitize=address,undefined)
    add_test(NAME store_cover_stream COMMAND store_cover_stream_tests)
    return()
endif()
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(STORE_COVER_DIAGNOSTICS "Private test-only artwork diagnostics" OFF)
add_subdirectory(../tls13/crypto crypto)
add_executable(tls_fixer_store cover.c json_stream.c http.c fetch.c vita.c
    ../tls13/provider/ca_index.c ../tls13/provider/platform_shims.c ../tls13/provider/user_runtime.c)
target_include_directories(tls_fixer_store PRIVATE ../tls13/provider ../tls13/crypto ../tls13/include)
target_compile_options(tls_fixer_store PRIVATE -std=gnu11 -Wall -Wextra -Werror -Os -fno-builtin -ffunction-sections -fdata-sections)
target_compile_definitions(tls_fixer_store PRIVATE STORE_COVER_DIAGNOSTICS=$<BOOL:${STORE_COVER_DIAGNOSTICS}> TLS_FIXER_ENABLE_DEBUG_LOG=0)
target_link_options(tls_fixer_store PRIVATE -nostdlib -Wl,--gc-sections -Wl,-e,module_start
    -Wl,--undefined=module_start -Wl,--undefined=module_stop
    -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/tls_fixer_store.map)
foreach(function malloc calloc realloc free memchr memcmp memcpy memmove memset strchr strcmp strlen strstr printf puts snprintf vsnprintf)
    target_link_options(tls_fixer_store PRIVATE "-Wl,--wrap=${function}")
endforeach()
target_link_libraries(tls_fixer_store PRIVATE mbedtls mbedx509 tfpsacrypto taihen_stub SceAppMgr_stub SceLibKernel_stub
    SceNet_stub SceRtc_stub SceIofilemgr_stub SceNpManager_stub SceSysmem_stub SceKernelModulemgr_stub
    SceKernelThreadMgr_stub SceProcessmgr_stub gcc)
vita_create_self(tls_fixer_store.suprx tls_fixer_store CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/module.yml UNSAFE)
