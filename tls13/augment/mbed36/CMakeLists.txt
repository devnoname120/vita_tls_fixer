cmake_minimum_required(VERSION 3.16)
project(tls_fixer_mbed36_control C)
option(TLS_FIXER_ENABLE_SANITIZERS "Enable host sanitizers" ON)
option(TLS_FIXER_NATIVE_HELLO_EXPERIMENT "Exercise the captured native offer" OFF)
if(TLS_FIXER_ENABLE_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_subdirectory(crypto)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(tls_fixer_core STATIC
    ${root}/src/cert_registry.c ${root}/src/cert_policy.c ${root}/src/error_map.c
    ${root}/src/module_gate.c ${root}/src/host_map.c ${root}/src/redirect_list.c
    ${root}/src/path.c ${root}/src/transport_path.c ${root}/src/registry.c ${root}/src/x509_names.c)
target_include_directories(tls_fixer_core PUBLIC ${root}/include)
target_compile_features(tls_fixer_core PUBLIC c_std_11)
set(TLS_FIXER_CRYPTO_PLATFORM_TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/test_platform.c")
set(TLS_FIXER_TRANSPORT_TEST_ARGUMENTS --include-rsa-cbc)
enable_testing()
add_subdirectory(${root}/tests/transport transport-tests)
if(TLS_FIXER_NATIVE_HELLO_EXPERIMENT)
    add_executable(native_clienthello_test native_clienthello_test.c ${TLS_FIXER_CRYPTO_PLATFORM_SOURCE})
    target_include_directories(native_clienthello_test PRIVATE ${root}/provider)
    target_link_libraries(native_clienthello_test PRIVATE mbedtls tls_fixer_vita_mock)
    target_compile_options(native_clienthello_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME native_clienthello COMMAND native_clienthello_test)
endif()
