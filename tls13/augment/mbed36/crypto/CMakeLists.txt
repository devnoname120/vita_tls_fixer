set(TLS_FIXER_MBEDTLS36_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/../../../../build/deps/mbedtls-3.6.7.tar.bz2"
    CACHE FILEPATH "Verified upstream Mbed TLS 3.6.7 release archive")
file(SHA256 "${TLS_FIXER_MBEDTLS36_ARCHIVE}" archive_hash)
if(NOT archive_hash STREQUAL "a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6")
    message(FATAL_ERROR "Mbed TLS 3.6.7 archive checksum mismatch")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_LIST_DIR}/user_config.h" CACHE FILEPATH "" FORCE)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
include_directories(BEFORE "${CMAKE_CURRENT_LIST_DIR}")
set(legacy_patch)
if(TLS_FIXER_NATIVE_HELLO_EXPERIMENT)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(legacy_patch PATCH_COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/apply_legacy_offer.py" <SOURCE_DIR>)
endif()
FetchContent_Declare(tls_fixer_mbedtls36 URL "${TLS_FIXER_MBEDTLS36_ARCHIVE}"
    URL_HASH SHA256=a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6 ${legacy_patch})
FetchContent_MakeAvailable(tls_fixer_mbedtls36)
add_library(tfpsacrypto ALIAS mbedcrypto)
if(TLS_FIXER_NATIVE_HELLO_EXPERIMENT)
    target_compile_definitions(mbedtls PRIVATE TLS_FIXER_DIAGNOSTIC_LEGACY_OFFERS=1)
endif()
target_compile_definitions(mbedx509 PRIVATE MBEDTLS_TEST_SW_INET_PTON)
foreach(target mbedtls mbedx509 mbedcrypto)
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
    target_compile_definitions(${target} PUBLIC TLS_FIXER_MBEDTLS36=1)
    target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections -O2)
endforeach()
set(TLS_FIXER_CRYPTO_PLATFORM_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../platform.c" PARENT_SCOPE)
