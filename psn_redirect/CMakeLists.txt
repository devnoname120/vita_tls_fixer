cmake_minimum_required(VERSION 3.16)
project(tls_fixer_psn C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(TLS_FIXER_KERNEL_STUB "${CMAKE_BINARY_DIR}/../kernel/stubs/libtls_fixer_kernel_stub.a" CACHE FILEPATH "Kernel redirect-list reader stub")
if(NOT EXISTS "${TLS_FIXER_KERNEL_STUB}")
    message(FATAL_ERROR "Build the tls_fixer kernel first to generate ${TLS_FIXER_KERNEL_STUB}")
endif()
add_executable(tls_fixer_psn main.c ../tls13/src/host_map.c ../tls13/src/redirect_list.c)
target_include_directories(tls_fixer_psn PRIVATE ../tls13/include)
target_compile_options(tls_fixer_psn PRIVATE -std=gnu11 -Wall -Wextra -Werror -Os -ffunction-sections -fdata-sections)
target_compile_definitions(tls_fixer_psn PRIVATE TLS_FIXER_ENABLE_DEBUG_LOG=0)
target_link_options(tls_fixer_psn PRIVATE -nostdlib -Wl,--gc-sections
    -Wl,--undefined=module_start -Wl,--undefined=module_stop)
target_link_libraries(tls_fixer_psn PRIVATE ${TLS_FIXER_KERNEL_STUB} taihen_stub SceLibKernel_stub gcc)
vita_create_self(tls_fixer_psn.suprx tls_fixer_psn CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/tls_fixer_psn.yml UNSAFE)
