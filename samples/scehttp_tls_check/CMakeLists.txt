cmake_minimum_required(VERSION 3.16)
project(vita_tls_check C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(TLS_CHECK_HOST "Build parser tests and a native screen renderer" OFF)
if(TLS_CHECK_HOST)
    enable_testing()
    add_executable(check_tests tests/test_model.c model.c)
    set(host_targets check_tests)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(FREETYPE QUIET IMPORTED_TARGET freetype2)
    endif()
    if(TARGET PkgConfig::FREETYPE)
        add_executable(check_preview tests/preview.c model.c ui.c font.c tests/font_host.c)
        target_link_libraries(check_preview PRIVATE PkgConfig::FREETYPE)
        list(APPEND host_targets check_preview)
    endif()
    foreach(target ${host_targets})
        target_include_directories(${target} PRIVATE .)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    endforeach()
    add_test(NAME check_model COMMAND check_tests)
else()
    if(NOT DEFINED VITASDK)
        set(VITASDK "$ENV{VITASDK}")
    endif()
    include("${VITASDK}/share/vita.cmake" REQUIRED)
    add_executable(vita_tls_check main.c model.c ui.c font.c font_vita.c)
    target_compile_options(vita_tls_check PRIVATE -Wall -Wextra -Werror -O2)
    target_link_options(vita_tls_check PRIVATE -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/vita_tls_check.map)
    target_link_libraries(vita_tls_check ScePvf_stub SceHttp_stub SceSsl_stub SceNet_stub SceNetCtl_stub
        SceSysmodule_stub SceDisplay_stub SceCtrl_stub SceLibKernel_stub
        SceKernelThreadMgr_stub SceIofilemgr_stub)
    vita_create_self(vita_tls_check.self vita_tls_check)
    vita_create_vpk(vita_tls_check.vpk TLSCK0001 vita_tls_check.self
        VERSION "01.00" NAME "Vita TLS Check")
endif()
