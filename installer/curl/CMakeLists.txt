cmake_minimum_required(VERSION 3.16)
project(tls_fixer_installer_curl C)

set(TLS_FIXER_CURL_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/deps/curl-8.17.0.tar.gz"
    CACHE FILEPATH "Checksum-pinned curl source archive")
if(NOT EXISTS "${TLS_FIXER_CURL_ARCHIVE}")
    message(FATAL_ERROR "Run python3 tools/fetch_curl.py on the host first")
endif()
file(SHA256 "${TLS_FIXER_CURL_ARCHIVE}" archive_sha256)
if(NOT archive_sha256 STREQUAL "e8e74cdeefe5fb78b3ae6e90cd542babf788fa9480029cfcee6fd9ced42b7910")
    message(FATAL_ERROR "curl source archive SHA-256 mismatch")
endif()

set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_CURL_MANUAL OFF CACHE BOOL "" FORCE)
set(HTTP_ONLY ON CACHE BOOL "" FORCE)
set(ENABLE_IPV6 OFF CACHE BOOL "" FORCE)
set(CURL_DISABLE_SOCKETPAIR ON CACHE BOOL "" FORCE)
set(HAVE_FCNTL_O_NONBLOCK OFF CACHE BOOL "" FORCE)
set(ENABLE_THREADED_RESOLVER OFF CACHE BOOL "" FORCE)
set(HAVE_PIPE2 "" CACHE INTERNAL "Vita has no pipe2 implementation" FORCE)
set(CURL_USE_OPENSSL ON CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_CA_BUNDLE "none" CACHE STRING "No default/system CA file" FORCE)
set(CURL_CA_PATH "none" CACHE STRING "No default/system CA directory" FORCE)
set(CURL_CA_FALLBACK OFF CACHE BOOL "" FORCE)

include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(installer_curl URL "${TLS_FIXER_CURL_ARCHIVE}"
    URL_HASH SHA256=e8e74cdeefe5fb78b3ae6e90cd542babf788fa9480029cfcee6fd9ced42b7910)
FetchContent_MakeAvailable(installer_curl)
# VitaSDK's OpenSSL archive has no console UI implementation.
target_compile_definitions(libcurl_static PRIVATE OPENSSL_NO_UI_CONSOLE)
